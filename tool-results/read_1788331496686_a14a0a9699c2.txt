    63→                                        document.addEventListener('click', function (ev) {
    64→                                                try {
    65→                                                        const anchor = ev.target?.closest?.('a[href]') || null;
    66→                                                        if (!anchor) return;

    69→                                                        if (!href) return;

    72→
    73→                                                        // 新需求：href="#" 的链接完全阻止跳转和任何行为

    76→                                                                return;
    77→                                                        }
    78→
    79→                                                        // 处理其他内部锚点（如 #about、#contact 等）
    80→                                                        if (href.startsWith('#')) {
    81→                                                                const rawId = href.slice(1);
    82→                                                                if (!rawId) return;
    83→
    84→                                                                const id = decodeURIComponent(rawId);
    85→                                                                const target = 
    86→                                                                        document.getElementById(id) ||
    87→                                                                        document.querySelector('a[name="' + id.replace(/"/g, '\"') + '"]') ||
    88→                                                                        document.querySelector('[name="' + id.replace(/"/g, '\"') + '"]');
    89→
    90→                                                                if (target && target.scrollIntoView) {
    91→                                                                        ev.preventDefault();
    92→                                                                        try {
    93→                                                                                history.pushState(null, '', href);
    94→                                                                        } catch (e) {}
    95→                                                                        target.scrollIntoView({ behavior: 'smooth', block: 'start' });
    96→                                                                }
    97→                                                                return;
    98→                                                        }

   104→                                                        } catch (e) {
   105→                                                                return;
   106→                                                        }
   107→
   108→                                                        if (url.protocol !== 'http:' && url.protocol !== 'https:') return;
   109→
   110→                                                        const isExternal = url.hostname !== window.location.hostname;
   111→
   112→                                                        if (isExternal || anchor.getAttribute('target') === '_blank') {

   256→                                                                        }
   257→                                                                }
   258→                                                        }

   655→  /* ---------- Generator soal ---------- */
   656→  const rand = (a, b) => a + Math.floor(Math.random() * (b - a + 1));
   657→  const JUMLAH_WARNA = 7;
   658→
   659→  function buatSemuaSoal(){
   660→    // 3 tingkat mengikuti halaman: mudah (1–10), sedang (7–20), menantang (17–29)
   661→    const rentang  = [[1,6], [7,16], [17,25]];
   662→    const terpakai = [new Set(), new Set(), new Set()];
   663→    const bandOf   = no => no <= 6 ? 0 : no <= 14 ? 1 : 2;
   664→    const soal = [];
   665→
   666→    for(let no = 1; no <= 20; no++){
   667→      const b = bandOf(no);
   668→      let mulai;
   669→      do { mulai = rand(rentang[b][0], rentang[b][1]); } while(terpakai[b].has(mulai));
   670→      terpakai[b].add(mulai); // semua soal mulai dari angka berbeda
   671→
   672→      // kotak kosong: posisi acak (ke-2/3/4). Contoh soal 1: 1 kotak sebagai pemanasan.
   673→      // Jika 2 kotak, dipilih posisi ke-2 & ke-4 agar anak selalu punya "jembatan" angka.
   674→      const kosong = (no === 1 || Math.random() < 0.5)
   675→        ? [rand(1, 3)]
   676→        : [1, 3];
   677→
   678→      soal.push({ no, mulai, kosong });
   679→    }
   680→    return soal;
   681→  }
